\begin{proposition}\label{prop:asym}
    Let \(A\) be a nonnegative matrix with spectral radius \(\rho(A)<1\). Then
    \((I-A)^{-1}=\sum_{k\ge0}A^k\).
\end{proposition}
\begin{proof}
    Neumann series; convergence follows from \(\rho(A)<1\). \qedhere
\end{proof}
