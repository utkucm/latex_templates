\begin{definition}\label{def:clearing}
    A \emph{clearing vector} is a fixed point \(p=\Phi(p)\) of the payment map
    \(\Phi:\R^n\to\R^n\).
\end{definition}

\begin{lemma}\label{lem:mono}
    \(\Phi\) is monotone: \(p\le q\) implies \(\Phi(p)\le\Phi(q)\).
\end{lemma}
\begin{proof}
    Each coordinate of \(\Phi\) is a minimum of nondecreasing affine functions,
    hence nondecreasing.
\end{proof}

\begin{theorem}[Eisenberg--Noe]\label{thm:en}
    Under \cref{lem:mono}, a clearing vector exists \citep{example2020}.
\end{theorem}
\begin{proof}[Proof of \cref{thm:en}]
    Apply Tarski's fixed-point theorem on the complete lattice \([0,\bar p]\).
\end{proof}

\begin{exercise}\label{ex:two}
    Compute the clearing vector for two banks that owe each other \(1\) and hold
    external assets \(a_1=a_2=\tfrac12\).
\end{exercise}
\begin{solution}
    By symmetry \(p_1=p_2=p\) with \(p=\min\{1,\tfrac12+p\}\), so \(p=1\) and
    \[
        p_1 = p_2 = 1 .
    \]
\end{solution}
