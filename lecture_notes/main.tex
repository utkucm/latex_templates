%% ============================================================================
%% lecture_notes.tex -- lecture-notes template with three versions.
%%
%% One source, three outputs, selected by \lecturemode:
%%   notes       my full notes: proofs, solutions, \todo notes     (default)
%%   handout     for students: proofs and solutions removed (blank space left
%%               where a proof was), no \todo, no teaching notes
%%   instructor  everything in notes plus teaching notes and timings, no \todo
%%
%% Select the mode ONE of two ways:
%%   (a) edit the defaults in section 1 below, or
%%   (b) override from the command line without touching the file (see Makefile):
%%         pdflatex "\def\lecturemode{handout}\input{lecture_notes}"
%%
%% Each default a mode sets can be overridden individually (section 1):
%%   \lectureproofs      auto | true | false   keep proof bodies
%%   \lecturesolutions   auto | true | false   keep solution bodies
%%   \lectureteach       auto | true | false   show teaching notes and timings
%%   \lecturetodos       auto | true | false   show \todo notes
%%   \lecturetitlepage   true | false          course title page
%%   \lecturetoc         true | false          table of contents
%%   \lectureproofspace  length left blank in place of a removed proof
%%
%% Lectures:   \newlecture[short title]{number}{topic}{date}
%% In lectures: theorem-like environments, exercise + solution, teachnote,
%%   \timing{minutes}, \todo{...}; proof as usual.
%% Do not define macros inside proof, solution or teachnote: their bodies are
%% discarded in some modes. Labels inside a discarded proof are discarded too,
%% so do not \cref a label that lives inside a proof.
%% ============================================================================

            \PackageError{lecturenotes}{Unknown mode `\lecturemode'}%
{Use \string\lecturemode = notes, handout, or instructor.}}}}

% \lec@override{<switch macro>}{<newif stem>}: true/false/auto
\newcommand{\lec@override}[2]{%
    \ifdefstring{#1}{true}{\csname lec@#2true\endcsname}{%
        \ifdefstring{#1}{false}{\csname lec@#2false\endcsname}{%
            \ifdefstring{#1}{auto}{}{%
                \PackageError{lecturenotes}{Bad value `#1' for \string#1}%
{Use true, false, or auto.}}}}}
\lec@override{\lectureproofs}{proofs}
\lec@override{\lecturesolutions}{solutions}

% >>> shared: references
% Citations are tracked during the current run, so this needs no .aux file from
% an earlier run. The bibliography is printed only if something was cited, so a
% document without citations has no empty "References" heading. A cite command
% that cannot be patched produces a warning instead of failing silently.
\usepackage{etoolbox}
\newif\ifhascitations
\newcommand{\trackcitations}[1]{%
    \pretocmd{#1}{\global\hascitationstrue}{}%
    {\PackageWarningNoLine{references}{Could not patch \string#1;
    citations made with it are not tracked}}%
}
\AtBeginDocument{%
    \trackcitations{\cite}\trackcitations{\nocite}%
    \trackcitations{\citet}\trackcitations{\citep}%
    \trackcitations{\citealt}\trackcitations{\citealp}%
    \trackcitations{\citeauthor}\trackcitations{\citeyear}\trackcitations{\citeyearpar}%
    \trackcitations{\Citet}\trackcitations{\Citep}%
    \trackcitations{\Citealt}\trackcitations{\Citealp}\trackcitations{\Citeauthor}%
}
% <<< shared: references

% Bibliography heading in the style of a lecture heading, with a TOC entry and
% a running header (natbib's default upper-cases the header and skips the TOC).
\renewcommand{\bibsection}{%
    \phantomsection
    \addcontentsline{toc}{section}{References}%
    \markboth{References}{}%
    {\noindent\Large\bfseries References\par}%
    \vspace{1.2em}%
}

% Bibliography at the end of the document, only if something was cited
\AtEndDocument{%
    \ifhascitations
    \newpage
    \bibliographystyle{apalike}%
    \bibliography{references}%
    \fi
}

% \newlecture[short title]{number}{topic}{date}
% Topic and date may be left empty. Starts a new page, typesets the heading,
% and adds the TOC entry and running header. The running header shows the
% optional short title if given, else the topic; if that would collide with
% the course name in the header, it falls back to "Lecture N".
\newlength{\lecturemarkwidth}
\newlength{\lectureavailwidth}
\NewDocumentCommand{\newlecture}{o m m m}{%
    \clearpage
    \setcounter{lecture}{\numexpr#2-1\relax}%
    \refstepcounter{lecture}% resets equation and theorem counters, anchors \label
    \ifblank{#3}%
    {\def\lecturetitle{Lecture \thelecture}}%
    {\def\lecturetitle{Lecture \thelecture: #3}}%
    \IfNoValueTF{#1}%
    {\let\lectureheader\lecturetitle}%
    {\ifblank{#1}%
        {\def\lectureheader{Lecture \thelecture}}%
    {\def\lectureheader{Lecture \thelecture: #1}}}%
    \settowidth{\lecturemarkwidth}{\small\lectureheader}%
    \settowidth{\lectureavailwidth}{\small\coursename}%
    \setlength{\lectureavailwidth}{\dimexpr\textwidth-\lectureavailwidth-2em\relax}%
    \ifdim\lecturemarkwidth>\lectureavailwidth
    \def\lectureheader{Lecture \thelecture}%
    \fi
    \addcontentsline{toc}{section}{\lecturetitle}%
    \markboth{\lectureheader}{}%
    {\noindent\Large\bfseries\lecturetitle\par}%
    \ifblank{#4}{}{\vspace{0.2em}{\noindent\small\itshape #4\par}}%
    \vspace{1.2em}%
}
