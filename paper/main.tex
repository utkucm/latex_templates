%% ============================================================================
%% paper.tex -- single-file research-paper template.
%%
%% One source, three versions, selected by \papermode:
%%   draft   author visible, 1.5 spacing, line numbers, \todo notes shown,
%%           "Draft <date>" in the footer                              (default)
%%   review  anonymous: no author, thanks, acknowledgments or PDF author;
%%           double spacing, line numbers, \todo notes hidden
%%   final   author visible, single spacing, no line numbers, no notes
%%
%% Select the mode ONE of two ways:
%%   (a) edit the defaults in section 1 below, or
%%   (b) override from the command line without touching the file (see Makefile):
%%         pdflatex "\def\papermode{review}\input{paper}"
%%
%% Each default a mode sets can be overridden individually (section 1):
%%   \paperspacing   auto | single | onehalf | double
%%   \paperlines     auto | true | false     line numbers
%%   \papernotes     auto | true | false     \todo notes visible
%%   \paperanon      auto | true | false     anonymize
%%   \paperappendix  true | false            include the appendix
%%   \paperbibstyle  BibTeX style (apalike; use aea/econometrica if installed)
%%   \paperbib       .bib file, without extension
%%
%% Macros and environments: \papersetup{...}, \makepaper, paperabstract,
%% acknowledgments, appendixmatter, \printreferences, \todo{...}.
%% Do not define macros inside acknowledgments or appendixmatter: they are
%% skipped when those parts are excluded.
%%
%% Anonymization covers what this file controls. It cannot see self-citations
%% ("in our earlier work"), file names, or author text in figures.
%% ============================================================================

% \paper@override{<switch macro>}{<newif stem>}: true/false/auto
\newcommand{\paper@override}[2]{%
    \ifdefstring{#1}{true}{\csname paper@#2true\endcsname}{%
        \ifdefstring{#1}{false}{\csname paper@#2false\endcsname}{%
            \ifdefstring{#1}{auto}{}{%
\PackageError{paper}{Bad value `#1' for \string#1}{Use true, false, or auto.}}}}}
